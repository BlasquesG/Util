// Carregar a coleção MOD16A2.061 para ET, LE, PET e PLE
var mod16a2Dataset = ee.ImageCollection('MODIS/061/MOD16A2')
                        .filter(ee.Filter.date('2000-01-01', '2024-06-30'));

// Selecionar as bandas ET, LE, PET, PLE
var et = mod16a2Dataset.select('ET');
var le = mod16a2Dataset.select('LE');
var pet = mod16a2Dataset.select('PET');
var ple = mod16a2Dataset.select('PLE');

// Calcular a média das bandas ET, LE, PET e PLE
var meanET = et.mean();
var meanLE = le.mean();
var meanPET = pet.mean();
var meanPLE = ple.mean();
